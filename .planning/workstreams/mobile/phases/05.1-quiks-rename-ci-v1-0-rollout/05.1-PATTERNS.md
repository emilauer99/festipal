### `packages/ui/src/theme.tsx` (new — provider/hook)

**Analog:** `apps/mobile/lib/fonts-context.tsx` (existing Context+Provider+hook triple for a
similar "resolve once at the root, consume via hook everywhere" concern — not fully read this
pass, but referenced directly by CONTEXT.md/RESEARCH.md as the sibling pattern: `FontsReadyProvider`
/ `useFontsReady`). Mirror its Provider-wraps-root + `useContext` hook shape.

**Core pattern to copy from RESEARCH.md Pattern 1** (already codebase-verified against
`react-native-website` official docs, adapted to this repo's alias-token contract):
```tsx
import { useColorScheme } from 'react-native';
import { colors, lightColors } from './tokens';

export function useTheme() {
  const scheme = useColorScheme(); // 'light' | 'dark' | null
  // hell-first (D-01): unresolved/null falls back to LIGHT, not dark.
  return scheme === 'dark' ? colors : lightColors;
}
```

**Consumer migration pattern (Pattern 2, RESEARCH.md)** — apply to all 17 token-consuming files,
using `FloatingNav.tsx`/`FestivalCard.tsx` as the concrete before/after reference:
```tsx
import { useMemo } from 'react';
import { StyleSheet } from 'react-native';
import { tokens } from '@quiks/ui';
import { useTheme } from '@quiks/ui'; // or apps/mobile local re-export, planner's discretion

function createStyles(colors: typeof tokens.colors) {
  return StyleSheet.create({
    card: { backgroundColor: colors.surfaceCard, borderColor: colors.borderSubtle },
  });
}

export function FestivalCard(/* ... */) {
  const colors = useTheme();
  const styles = useMemo(() => createStyles(colors), [colors]);
  // ...
}
```
